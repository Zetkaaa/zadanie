#include <iostream>
using namespace std;

int main()
{
    int a = 0;
    int b = 0;

    cout << "Podaj dwie liczby: ";
    cin >> a >> b;

    if (a == b)
    {
        cout << a<<"="<<b << endl;
    }
    else if(a > b)
    {
        cout <<a<<">"<<b<<endl;
    }
    else if(a < b)
    {
        cout<<a<<"<"<<b<<endl;
    }
    return 0;
}
